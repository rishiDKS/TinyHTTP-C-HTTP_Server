cmake_minimum_required(VERSION 3.16)
project(TinyHTTP VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(tinyhttp src/main.cpp src/server.cpp src/http.cpp src/thread_pool.cpp)
target_include_directories(tinyhttp PRIVATE include)
target_link_libraries(tinyhttp PRIVATE Threads::Threads)
target_compile_options(tinyhttp PRIVATE -Wall -Wextra -Wpedantic)
